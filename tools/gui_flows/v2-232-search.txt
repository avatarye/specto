# v2 #232: type-to-search in the file tree. The tree starts collapsed (#231), so the
# readmes two and three folders down are not listed; typing in the search box finds them
# by name anyway: the hits replace the tree, best first (exact name, then prefix, then
# the rest; shallower first among equals), the footer counts them. A click picks a hit
# (selected_files carries it), Ctrl adds another; a word nothing matches says which
# backend looked; Escape returns to the tree with the picks kept; the reveal button on a
# hit goes back to the tree at that file, its folders listed and opened, the row selected.
# The session runs with SPECTO_SEARCH_BACKEND=scan, so the flow is the same on a machine
# with Everything as on one without it. Project: tools/gui_flows/v2-232-seed.py <folder>.
resize 1600 900
wait 1500
wait_until treeRow:@file_tree:notes 20000
assert_found treeRow:@file_tree:docs
assert_missing treeRow:@file_tree:notes/readme.md
assert_missing treeRow:@file_tree:docs/deep/er/readme-draft.md
shot v2-232-01-collapsed-tree.png
# typing finds files the tree has not listed
click treeSearch:@file_tree
type readme
wait_until searchHit:@file_tree:docs/deep/er/readme-draft.md 10000
assert_host @file_tree search.backend "scan"
assert_host @file_tree search.hits ["notes/readme.md","docs/readme.txt","docs/deep/er/readme-draft.md"]
assert_host @file_tree search.more false
assert_found searchHitFolder:@file_tree:docs/deep/er/readme-draft.md
assert_text searchHitFolder:@file_tree:docs/deep/er/readme-draft.md docs/deep/er
assert_missing fileTreeList:@file_tree
assert_text fileTreeCaption:@file_tree 3 hits
shot v2-232-02-hits-replace-the-tree.png
# a click picks the hit; Ctrl adds one
click searchHitName:@file_tree:docs/deep/er/readme-draft.md 0.3 0.5
assert_host @file_tree selected ["docs/deep/er/readme-draft.md"] 5000
ctrlclick searchHitName:@file_tree:notes/readme.md 0.3 0.5
assert_host @file_tree selected_rows ["docs/deep/er/readme-draft.md","notes/readme.md"] 5000
shot v2-232-03-two-hits-picked.png
# a narrower word: every word must match the name
click treeSearch:@file_tree
key ctrl+a
type readme draft
wait_until searchHitName:@file_tree:docs/deep/er/readme-draft.md 10000
assert_host @file_tree search.hits ["docs/deep/er/readme-draft.md"]
assert_text fileTreeCaption:@file_tree 1 hit
# a miss says which backend looked
key ctrl+a
type zzz
wait_until fileTreeSearchEmpty:@file_tree 10000
assert_host @file_tree search.hits []
assert_text fileTreeSearchEmpty:@file_tree no matches
assert_text fileTreeSearchEmpty:@file_tree a scan of the folder
shot v2-232-04-no-matches-names-the-backend.png
# Escape: back to the tree, the search cleared, the picks kept
key escape
wait 500
assert_host @file_tree search.text ""
assert_found fileTreeList:@file_tree
assert_missing fileTreeHits:@file_tree
assert_host @file_tree selected_rows ["docs/deep/er/readme-draft.md","notes/readme.md"]
shot v2-232-05-escape-back-to-the-tree.png
# reveal: the tree at the hit, its folders listed and open, the row selected
click treeSearch:@file_tree
type draft
wait_until searchHit:@file_tree:docs/deep/er/readme-draft.md 10000
hover searchHit:@file_tree:docs/deep/er/readme-draft.md 0.5 0.5
wait 300
assert_found searchHitReveal:@file_tree:docs/deep/er/readme-draft.md
click searchHitReveal:@file_tree:docs/deep/er/readme-draft.md
wait_until treeRow:@file_tree:docs/deep/er/readme-draft.md 10000
assert_host @file_tree search.text ""
assert_host @file_tree expanded ["docs","docs/deep","docs/deep/er"]
assert_host @file_tree selected_rows ["docs/deep/er/readme-draft.md"]
assert_host @file_tree selected ["docs/deep/er/readme-draft.md"]
assert_missing treeRow:@file_tree:notes/readme.md
shot v2-232-06-revealed-in-the-tree.png
