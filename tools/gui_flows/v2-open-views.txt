# v2 SPEC-32: what a double-click on a file in the file tree opens. A double-click opens the
# file's dedicated view: a view of the type that shows its kind, fed that one file by a
# pinned link, labelled "<name> · <folder>", left alone by later picks; the same file again
# reveals that view instead of adding another; a file no view shows opens the "Not
# supported" text view. Ctrl+double-click opens the contextual preview on the tree's
# picks, tagged "Contextual · follows selection" in its header and on its overworld card,
# and the file ends up selected. The context menu offers both. Folders still toggle.
# Project: tools/gui_flows/v2-122-seed.py <folder>.
resize 1600 960
wait 1500
assert_found slot:@file_tree
assert_missing slot:@preview
wait_until treeRow:@file_tree:docs 20000
expand @file_tree:docs
expand @file_tree:images
expand @file_tree:papers
wait_until treeRow:@file_tree:docs/plan.md 20000
wait_until treeRow:@file_tree:papers/report.pdf 10000
# 1 double-click: a dedicated markdown view of the note, labelled name · folder, no tag
assert_missing slot:@markdown
dblclick treeRow:@file_tree:docs/plan.md 0.6 0.5
wait_until slot:@markdown 15000
wait 1500
assert_text title:@markdown plan.md · docs
assert_missing contextual:@markdown
assert_missing slot:@preview
shot open-views-01-double-click-opens-a-dedicated-view.png
print snap {"target": "@markdown"}
# 2 a later single click elsewhere does not change it
click treeRow:@file_tree:images/00001.jpg 0.6 0.5
wait 1000
assert_text title:@markdown plan.md · docs
# 3 the same file again: the view is revealed, never a second one
dblclick treeRow:@file_tree:docs/plan.md 0.6 0.5
wait 1200
assert_missing slot:@markdown#2
assert_found slot:@markdown
# 4 a file no view shows: the "Not supported" text view, labelled the same way
dblclick treeRow:@file_tree:odd.xyz 0.6 0.5
wait_until slot:@text 15000
wait 800
assert_text text:@text:body Not supported: odd.xyz (.xyz)
assert_text title:@text odd.xyz
shot open-views-02-not-supported.png
# 5 a double-click on a folder still toggles it
mark treeRow:@file_tree:papers/report.pdf
dblclick treeRow:@file_tree:papers 0.6 0.5
wait 800
assert_missing treeRow:@file_tree:papers/report.pdf
dblclick treeRow:@file_tree:papers 0.6 0.5
wait 800
assert_found treeRow:@file_tree:papers/report.pdf
# 6 Ctrl+double-click: the contextual preview on the picks, tagged, the file selected
# (the "Not supported" view closed first, so the preview lands on screen below the note)
click close:@text
wait 800
assert_missing slot:@text
ctrldblclick treeRow:@file_tree:images/00001.jpg 0.6 0.5
wait_until slot:@preview 15000
wait 1500
assert_found contextual:@preview
assert_text contextual:@preview Contextual
assert_text follows:@preview follows selection
assert_missing preview:@preview:card
assert_missing slot:@preview#2
shot open-views-03-ctrl-double-click-opens-the-contextual-preview.png
print snap {"target": "@file_tree"}
# 7 it follows the next pick (the pdf), the dedicated note view stays as it was
click treeRow:@file_tree:papers/report.pdf 0.6 0.5
wait 1500
assert_missing preview:@preview:card
assert_text title:@markdown plan.md · docs
print snap {"target": "@preview"}
# 8 the overworld card wears the tag too
key tab
wait_until overCard:@preview 8000
wait 600
assert_found overContextual:@preview
assert_missing overContextual:@markdown
shot open-views-04-overworld-contextual-tag.png
key tab
wait 1200
wait_until treeRow:@file_tree:docs/README.md 8000
# 9 the context menu on a file offers both ways: the contextual one re-points the
# preview (the file picked), the dedicated one opens the file's dedicated view
rclick treeRow:@file_tree:docs/README.md 0.6 0.5
wait_until menu:file_tree.open_dedicated 8000
assert_found menu:file_tree.open_contextual
shot open-views-05-context-menu-open-entries.png
click menu:file_tree.open_contextual
wait 1500
assert_missing slot:@preview#2
assert_missing preview:@preview:card
print snap {"target": "@file_tree"}
click close:@preview
wait 800
assert_missing slot:@preview
rclick treeRow:@file_tree:docs/README.md 0.6 0.5
wait_until menu:file_tree.open_dedicated 8000
click menu:file_tree.open_dedicated
wait_until slot:@markdown#2 15000
wait 800
assert_text title:@markdown#2 README.md · docs
assert_missing contextual:@markdown#2
