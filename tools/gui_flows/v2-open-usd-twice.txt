# v2 SPEC-34: two stages double-clicked in turn. The first opens its dedicated usd view;
# the second opens its own usd view too (one live RTX view per canvas, ADR 0025), and the
# first collapses to its title bar to make room, in the same undo step, with a notice
# naming it. Expanding the first collapses the second; Undo brings the second back.
# Project: tools/gui_flows/v2-open-usd-seed.py <folder>. The suite runs the stub USD
# source (no renderer), so no picture is judged here.
resize 1600 960
wait 1500
assert_found slot:@file_tree
wait_until treeRow:@file_tree:worlds 20000
expand @file_tree:worlds
expand @file_tree:props
wait_until treeRow:@file_tree:worlds/yard.usda 15000
wait_until treeRow:@file_tree:props/crate.usda 15000
# 1 the first stage: its dedicated usd view, live
dblclick treeRow:@file_tree:worlds/yard.usda 0.6 0.5
wait_until slot:@usd 15000
wait 1200
assert_text title:@usd yard.usda · worlds
print snap {"target": "@usd"}
# 2 the second stage: its own usd view; the first collapses to make room, said in a notice
dblclick treeRow:@file_tree:props/crate.usda 0.6 0.5
wait_until slot:@usd#2 15000
wait 1200
assert_missing notice:error
assert_text title:@usd#2 crate.usda · props
assert_text noticeText:info yard.usda · worlds collapsed to make room for crate.usda · props
shot open-usd-twice-01-second-stage-collapses-the-first.png
print snap {"target": "canvas", "layout": true}
# 3 expanding the first stage's view collapses the second
click collapse:@usd
wait 1200
assert_missing notice:error
assert_text noticeText:info crate.usda · props collapsed to make room for yard.usda · worlds
shot open-usd-twice-02-expand-the-first-collapses-the-second.png
print snap {"target": "canvas", "layout": true}
